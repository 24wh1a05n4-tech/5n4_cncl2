  double height = 5.4;

    print('$name is an adult.');
  } else {
    print('$name is not an adult.');
  }

  greet(name);

  for (int i = 1; i <= 5; i++) {
    print('Loop iteration $i');
  }

  List<String> friends = ['Divya', 'Kavya', 'Priya', 'Sravani', 'Nandini'];

  for (String friend in friends) {
    print('Hello, $friend!');
  }

  Person person = Person(name, age, height);
  person.introduce();
}

void greet(String name) {
  print('Welcome to Dart, $name!');
}

class Person {
  String name;
  int age;
  double height;

  Person(this.name, this.age, this.height);

  void introduce() {
    print('My name is $name. I am $age years old and my height is $height feet.');
  }
}

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      debugShowCheckedModeBanner: false,
      home: ResponsivePage(),
    );
  }
}

class ResponsivePage extends StatelessWidget {
  const ResponsivePage({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Simple Responsive UI')),
      body: LayoutBuilder(
        builder: (context, constraints) {
          // Calculate the height needed for the column content
          // Two images (300 + 300) = 600 vertical pixels
          final isSmallWidth = constraints.maxWidth < 600;
          final isSmallHeight = constraints.maxHeight < 600;

          // Render a scrollable view if the screen is too small in either direction
          if (isSmallWidth || isSmallHeight) {
            return SingleChildScrollView(
              child: Center(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Image.asset(
                      'assets/images/download (1).jpeg',
                      width: 300,
                      height: 300,
                      fit: BoxFit.cover,
                    ),
                    Image.asset(
                      'assets/images/download.jpeg',
                      width: 300,
                      height: 300,
                      fit: BoxFit.cover,
                    ),
                  ],
                ),
              ),
            );
          } else {
            // Large screen: center Row layout
            return Center(
              child: Row(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Image.asset(
                    'assets/images/download (1).jpeg',
                    width: 300,
                    height: 300,
                    fit: BoxFit.cover,
                  ),
                  Image.asset(
                    'assets/images/download.jpeg',
                    width: 300,
                    height: 300,
                    fit: BoxFit.cover,
                  ),
                ],
              ),
            );
          }
        },
      ),
    );
  }

  Widget myBox(String text, Color color) {
    return Container(
      height: 150,
      width: 150,
      margin: const EdgeInsets.all(10),
      color: color,
      child: Center(
        child: Text(text, style: const TextStyle(color: Colors.white)),
      ),
    );
  }
}

import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      debugShowCheckedModeBanner: false,
      home: HomePage(),
    );
  }
}

class HomePage extends StatelessWidget {
  const HomePage({super.key});

  @override
  Widget build(BuildContext context) {
    // Get screen size and orientation
    final Size screenSize = MediaQuery.sizeOf(context);
    final Orientation orientation = MediaQuery.orientationOf(context);
    final double width = screenSize.width;

    // Determine view type based on width
    final String viewType;
    if (width < 600) {
      viewType = 'Mobile View';
    } else if (width < 1024) {
      viewType = 'Tablet View';
    } else {
      viewType = 'Desktop View';
    }

    return Scaffold(
      appBar: AppBar(title: const Text('MediaQuery + Breakpoints')),
      body: Container(
        color: Colors.lightBlue,
        child: Center(
          child: Text(
            'Responsive Layout Example\n\n'
            'View Type: $viewType\n'
            'Screen Width: ${width.toStringAsFixed(2)}\n'
            'Screen Height: ${screenSize.height.toStringAsFixed(2)}\n'
            'Orientation: $orientation',
            textAlign: TextAlign.center,
            style: const TextStyle(color: Colors.white, fontSize: 20),
          ),
        ),
      ),
    );
  }
}

        '/': (context) => const FirstRoute(), // First route (home)
        '/second': (context) => const SecondRoute(), // Second route
        '/third': (context) => const ThirdRoute(), // Third route
      },
    );
  }
}

import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

// Root is StatelessWidget
class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      debugShowCheckedModeBanner: false,
      home: CounterPage(),
    );
  }
}

// CounterPage is StatefulWidget
class CounterPage extends StatefulWidget {
  const CounterPage({super.key});

  @override
  State<CounterPage> createState() => _CounterPageState();
}

class _CounterPageState extends State<CounterPage> {
  int counter = 0;

  void increment() {
    setState(() {
      counter++;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("Stateless + Stateful Counter - 24WH1A05M4"),
        backgroundColor: Colors.blue,
        foregroundColor: Colors.white,
      ),
      body: Center(
        child: Text(
          "Counter: $counter",
          style: const TextStyle(fontSize: 24),
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: increment,
        child: const Icon(Icons.add),
      ),
    );
  }
}




5b) Implement state management using set State and Provider.

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

void main() {
  runApp(
    ChangeNotifierProvider(
      create: (_) => CounterModel(),
      child: const MyApp(),
    ),
  );
}

// Root is StatelessWidget
class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      debugShowCheckedModeBanner: false,
      home: CounterPage(),
    );
  }
}

// Provider class (state management)
class CounterModel extends ChangeNotifier {
  int counter = 0;

  void increment() {
    counter++;
    notifyListeners(); // notifies UI to rebuild
  }
}

// CounterPage (StatelessWidget, listens to Provider)
class CounterPage extends StatelessWidget {
  const CounterPage({super.key});

  @override
  Widget build(BuildContext context) {
    final counter = context.watch<CounterModel>().counter;

    return Scaffold(
      appBar: AppBar(
        title: const Text("Provider Counter"),
        backgroundColor: Colors.blue,
        foregroundColor: Colors.white,
      ),
      body: Center(
        child: Text(
          "Counter: $counter",
          style: const TextStyle(fontSize: 24),
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () => context.read<CounterModel>().increment(),
        child: const Icon(Icons.add),
      ),
    );
  }
}
